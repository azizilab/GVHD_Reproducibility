create_cluster_subset <- function(dt, basis) {
  basis <- as.character(basis)
  threshold <- new_thresholds[basis]
  subset_dt <- dt[dt[[basis]] > threshold,
                  .(CDR3, Vgene, Jgene, gene = "TRB", score = get(basis), PatientID)]
  return(unique(subset_dt))
}

dt_b <- vector("list", 10)
for (i in 1:10) {
  dt_b[[i]] <- create_cluster_subset(dt, i)
}
calculate_hydrophobicity <- function(cdr3_string, hydrophobicity_table) {
  cdr3_string <- as.character(cdr3_string)
  vals <- hydrophobicity_table$Hydrophobicity[
    match(strsplit(cdr3_string, NULL)[[1]], hydrophobicity_table$AA)
  ]
  vals <- vals[!is.na(vals)]
  if (length(vals) > 0) mean(vals) else NA
}

calculate_hydrophobicity_for_positions <- function(strings, hydrophobicity_table) {
  sapply(strings, function(string) {
    string <- as.character(string)
    if (nchar(string) < 13) return(NA)
    calculate_hydrophobicity(substring(string, 5, 13), hydrophobicity_table)
  })
}

  b1 = calculate_hydrophobicity_for_positions(dt_b[[1]]$CDR3, hydrophobicity_chart),

# Pairwise Wilcoxon p-values (Holm-adjusted)
groups <- levels(combined_hydrophobicity_grouped$BasisGroup)
pairs  <- combn(groups, 2, simplify = FALSE)
pwilcox <- sapply(pairs, function(g) {
  x <- combined_hydrophobicity_grouped$Hydrophobicity[combined_hydrophobicity_grouped$BasisGroup == g[1]]
  y <- combined_hydrophobicity_grouped$Hydrophobicity[combined_hydrophobicity_grouped$BasisGroup == g[2]]
  wilcox.test(x, y)$p.value
})
p_adj <- p.adjust(pwilcox, method = "holm")

# Summary statistics per group
hydro_summary <- combined_hydrophobicity_grouped %>%
  group_by(Group = BasisGroup) %>%
  summarize(
    N      = n(),
    Median = round(median(Hydrophobicity, na.rm = TRUE), 3),
    SD     = round(sd(Hydrophobicity,     na.rm = TRUE), 3),
    .groups = "drop"
  )

# Pairwise comparison table with % difference in medians
medians <- setNames(hydro_summary$Median, hydro_summary$Group)
hydro_pairs <- data.frame(
  Comparison       = sapply(pairs, function(g) paste(g[1], "vs", g[2])),
  Median_G1        = sapply(pairs, function(g) medians[g[1]]),
  Median_G2        = sapply(pairs, function(g) medians[g[2]]),
  Pct_Diff         = sapply(pairs, function(g) {
    round((medians[g[2]] - medians[g[1]]) / abs(medians[g[1]]) * 100, 1)
  }),
  P_value          = signif(pwilcox, 3),
  P_adj_Holm       = signif(p_adj,   3)
)

Running Hydrophobicity on ALLOREACTIVE CLONES

allclones <- fread(
  "Inputs/AllClones_all.csv",
  select = c("cdr3_amino_acid", "v_resolved", "j_resolved", "type", "GVHD_GRADE_1", "GVHD_GRADE_2", "PatientID")
)
setnames(allclones, c("cdr3_amino_acid", "v_resolved", "j_resolved"), c("CDR3", "Vgene", "Jgene"))

alloreactive_clones <- allclones %>%
  filter(type == "allo") %>%
  distinct(CDR3, Vgene, Jgene, .keep_all = TRUE)

alloreactive_clones$Hydrophobicity_5_13 <- calculate_hydrophobicity_for_positions(
  alloreactive_clones$CDR3,
  hydrophobicity_table = hydrophobicity_chart
)
grades_to_keep <- c("No", "Mild", "Severe")

alloreactive_clones_clean <- alloreactive_clones %>%
  filter(!is.na(Hydrophobicity_5_13)) %>%
  filter(GVHD_GRADE_1 %in% grades_to_keep) %>%
  mutate(GVHD_GRADE_1 = factor(GVHD_GRADE_1, levels = grades_to_keep))

# Define pairwise comparisons (Wilcoxon)
comparisons <- combn(grades_to_keep, 2, simplify = FALSE)

pp2 <- ggplot(alloreactive_clones_clean,
              aes(x = GVHD_GRADE_1, y = Hydrophobicity_5_13, fill = GVHD_GRADE_1)) +
  geom_violin(width = 0.35, trim = FALSE) +
  geom_boxplot(width = 0.15, outlier.shape = NA,
               color = "black") +
  labs(
    title = "Hydrophobicity (AA 5-13) in Alloreactive Clones\nby GVHD Grade",
    x = "GVHD Grade",
    y = "Hydrophobicity Score"
  ) +
  theme_pubr() +
  scale_fill_manual(values = c(
    "No"     = "#09BB8C",
    "Mild"   = "#005A8F",
    "Severe" = "#B85000"
  )) +
  theme(
    plot.title       = element_text(size = 8,  hjust = 0.5),
    axis.title.x     = element_text(size = 7),
    axis.title.y     = element_text(size = 7),
    axis.text        = element_text(size = 6),
    legend.position  = "none",
    axis.ticks.length = unit(0.01, "cm")
  ) +
  stat_compare_means(
    method = "wilcox.test",
    comparisons = comparisons,
    label = "p.signif",
    p.adjust.method = "holm",
    size = 3
  )

print(pp2)
